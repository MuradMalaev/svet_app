import { icon, esc, $, toast, subbar, progressBar, ringSvg, confirmDialog, plural, clamp } from '../core/ui.js'; import { getPrayed, togglePrayed, getKhatm, setKhatm, getAzkarToday, getMemorized, getNamesViewed } from '../core/store.js'; import { getAzkar } from '../core/data.js'; import { dayKey, addDays, startOfDay, diffDays, parseDay, fmtDMY } from '../core/time.js'; import { OBLIGATORY, QUOTES, TOTAL_PAGES, AZKAR_CATS } from '../core/meta.js'; import { scheduleFor } from './home.js'; /* ═════ Трекер ═════ */ export async function renderTracker(view) { const today = dayKey(); let azkarTotal = 0; try { const z = await getAzkar(); azkarTotal = AZKAR_CATS.reduce((t, c) => t + (z[c.id]?.length || 0), 0); } catch { /* без данных азкаров */ } const draw = () => { const prayed = getPrayed(today), cnt = OBLIGATORY.filter((p) => prayed[p.key]).length; const sch = scheduleFor(0), now = Date.now(); const nxt = OBLIGATORY.find((p) => sch.ms[p.key] && sch.ms[p.key] > now); const k = getKhatm(), az = Object.keys(getAzkarToday().done).length, mem = getMemorized().size, nm = getNamesViewed().size; const quote = QUOTES[Math.floor((startOfDay() - new Date(2024, 0, 1)) / 86400000) % QUOTES.length]; // последние 7 дней — из истории отметок const hist = Array.from({ length: 7 }, (_, i) => { const d = addDays(startOfDay(), i - 6), p = getPrayed(dayKey(d)); return { d, n: OBLIGATORY.filter((o) => p[o.key]).length }; }); const cats = [ { ic: 'quran', t: 'Коран', v: k.active ? `${k.page} / ${TOTAL_PAGES}` : 'план не задан', f: k.active ? k.page / TOTAL_PAGES : 0, href: '#/tracker/khatm' }, { ic: 'azkar', t: 'Азкары', v: `${az} / ${azkarTotal}`, f: azkarTotal ? az / azkarTotal : 0, href: '#/azkar' }, { ic: 'hifz', t: 'Хифз', v: `${mem} / 114 сур`, f: mem / 114, href: '#/hifz' }, { ic: 'names', t: '99 имён', v: `${nm} / 99`, f: nm / 99, href: '#/names' }, ]; view.innerHTML = `${subbar('Трекер', '#/')}
${icon('mosque')}
Сегодня ${cnt} / 5 намазов${nxt ? 'Следующий: ' + nxt.ru : 'Следующий: Фаджр (завтра)'}
${OBLIGATORY.map((p) => ``).join('')}
${cats.map((c) => `
${icon(c.ic)}${c.t}
${esc(c.v)}${progressBar(c.f, c.t)}`).join('')}

Намазы за 7 дней

${hist.map((h) => `
${h.n}${['Вс', 'Пн', 'Вт', 'Ср', 'Чт', 'Пт', 'Сб'][h.d.getDay()]}
`).join('')}

${esc(quote)}

`; }; draw(); view.onclick = (e) => { const b = e.target.closest('[data-p]'); if (!b) return; togglePrayed(today, b.dataset.p); draw(); }; return () => { view.onclick = null; }; } /* ═════ Хатм-план ═════ */ const effPerDay = (k) => { const remaining = clamp(TOTAL_PAGES - k.page + 1, 0, TOTAL_PAGES); if (k.mode === 0 || !k.target) return k.pagesPerDay; const left = diffDays(startOfDay(), parseDay(k.target)); if (left <= 0) return remaining; return clamp(Math.ceil(remaining / left), 1, TOTAL_PAGES); }; const finishDate = (k) => { if (k.mode === 1 && k.target) return parseDay(k.target); const remaining = clamp(TOTAL_PAGES - k.page + 1, 0, TOTAL_PAGES); return addDays(startOfDay(), Math.ceil(remaining / k.pagesPerDay)); }; export function renderKhatm(view) { let setup = { mode: 0, pages: 5, target: dayKey(addDays(startOfDay(), 30)) }; const draw = () => { const k = getKhatm(); if (!k.active) return drawSetup(); const remaining = clamp(TOTAL_PAGES - k.page + 1, 0, TOTAL_PAGES); const frac = clamp((k.page - 1) / TOTAL_PAGES, 0, 1), fd = finishDate(k); view.innerHTML = `${subbar('Хатм-план', '#/tracker')}
${ringSvg(frac, Math.round(frac * 100) + '%')}
Осталось ${remaining} ${plural(remaining, 'страница', 'страницы', 'страниц')} Читай ${effPerDay(k)} стр/день, чтобы закончить к ${fmtDMY(fd)}

План сам пересчитает темп, если ты отстал или прочитал больше

`; }; const drawSetup = () => { const days = Math.ceil(TOTAL_PAGES / setup.pages), fd = addDays(startOfDay(), days); const left = Math.max(1, diffDays(startOfDay(), parseDay(setup.target))); view.innerHTML = `${subbar('Хатм-план', '#/tracker')}

Выберите, как планировать чтение всего Корана целиком (${TOTAL_PAGES} страниц Мединского мусхафа):

${setup.mode === 0 ? `

Ты закончишь весь Коран примерно через ${days} ${plural(days, 'день', 'дня', 'дней')} — к ${fmtDMY(fd)}

` : `

Понадобится читать примерно ${Math.ceil(TOTAL_PAGES / left)} стр. в день

`}
`; }; draw(); view.onclick = async (e) => { const m = e.target.closest('[data-m]'); if (m) { setup.mode = +m.dataset.m; return drawSetup(); } if (e.target.closest('#k-start')) { const k = getKhatm(); if (setup.mode === 1 && diffDays(startOfDay(), parseDay(setup.target)) < 1) return toast('Выберите дату в будущем'); setKhatm({ active: true, mode: setup.mode, pagesPerDay: setup.pages, target: setup.mode === 1 ? setup.target : null, start: dayKey(), page: k.page || 1 }); return draw(); } if (e.target.closest('#k-save')) { const v = Math.round(+$('#kp', view).value); if (!v || v < 1 || v > TOTAL_PAGES) return toast(`Введите страницу от 1 до ${TOTAL_PAGES}`); setKhatm({ ...getKhatm(), page: v }); toast('Страница сохранена'); return draw(); } if (e.target.closest('#k-edit')) { if (await confirmDialog('Изменить план?', 'Текущая страница сохранится — вы выберете новый темп или дату.', 'Изменить')) { setKhatm({ ...getKhatm(), active: false }); draw(); } } }; view.oninput = (e) => { if (e.target.id === 'kpd') { setup.pages = +e.target.value; const days = Math.ceil(TOTAL_PAGES / setup.pages); $('#kpd-v', view).textContent = setup.pages; $('#k-hint', view).textContent = `Ты закончишь весь Коран примерно через ${days} ${plural(days, 'день', 'дня', 'дней')} — к ${fmtDMY(addDays(startOfDay(), days))}`; } if (e.target.id === 'kd') { setup.target = e.target.value || setup.target; const left = Math.max(1, diffDays(startOfDay(), parseDay(setup.target))); $('#k-hint', view).textContent = `Понадобится читать примерно ${Math.ceil(TOTAL_PAGES / left)} стр. в день`; } }; return () => { view.onclick = null; view.oninput = null; }; }
